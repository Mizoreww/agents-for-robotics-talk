const RADAR_RELATIVE_FLOOR = 30;
const RADAR_RELATIVE_CEILING = 70;

function relativeRadarScore(value, start, end) {
  const progress =
    start === end
      ? 1
      : Math.max(0, Math.min(1, (value - start) / (end - start)));
  return (
    RADAR_RELATIVE_FLOOR +
    (RADAR_RELATIVE_CEILING - RADAR_RELATIVE_FLOOR) * progress
  );
}

function positionHeroOrbit() {
  if (!heroOrbit || !heroIntro || window.innerWidth <= 1180) {
    heroOrbit?.style.removeProperty("top");
    heroOrbit?.style.removeProperty("left");
    return;
  }

  const introRect = heroIntro.getBoundingClientRect();
  const orbitRect = heroOrbit.getBoundingClientRect();
  const leftGap = Math.min(80, Math.max(50, window.innerWidth * 0.06));
  const top = heroOrbit.offsetTop + introRect.bottom + 8 - orbitRect.bottom;
  const desiredLeft = introRect.left - orbitRect.width - leftGap;
  const left = heroOrbit.offsetLeft + desiredLeft - orbitRect.left;
  heroOrbit.style.top = `${Math.round(top)}px`;
  heroOrbit.style.left = `${Math.round(left)}px`;
}

function scheduleHeroOrbitPosition() {
  window.requestAnimationFrame(() =>
    window.requestAnimationFrame(positionHeroOrbit),
  );
}

let engineeringMorphFrame = 0;

function updateEngineeringMorph() {
  engineeringMorphFrame = 0;
  if (
    !engineeringMorph ||
    !engineeringComparison ||
    !engineeringBefore ||
    !engineeringAfter
  ) {
    return;
  }

  const comparisonWidth = engineeringComparison.clientWidth;
  const compact =
    window.innerWidth <= 900 ||
    comparisonWidth < 1000 ||
    reduceMotion.matches;
  if (compact) {
    engineeringMorph.classList.remove("is-morph-ready");
    [
      "--comparison-card-height",
      "--comparison-stage-height",
      "--comparison-before-width",
      "--comparison-after-width",
      "--comparison-arrow-center",
      "--comparison-inline-height",
      "--comparison-before-opacity",
      "--comparison-arrow-opacity",
    ].forEach((property) => engineeringMorph.style.removeProperty(property));
    return;
  }

  engineeringMorph.classList.add("is-morph-ready");
  const columnGap = Math.min(88, Math.max(68, comparisonWidth * 0.065));
  const availableWidth = Math.max(0, comparisonWidth - columnGap);
  const beforeWidth = availableWidth / 2;
  const afterWidth = availableWidth - beforeWidth;
  engineeringMorph.style.setProperty(
    "--comparison-before-width",
    `${beforeWidth.toFixed(2)}px`,
  );
  engineeringMorph.style.setProperty(
    "--comparison-arrow-center",
    `${(beforeWidth + columnGap / 2).toFixed(2)}px`,
  );

  const beforeVideo = engineeringBefore.querySelector(":scope > video");
  const finalInlineHeight =
    beforeVideo?.offsetHeight || beforeWidth * (9 / 16);
  const stickyTop =
    Number.parseFloat(getComputedStyle(engineeringComparison).top) || 96;
  const initialVideoHeight = 430;
  const startTop = window.innerHeight / 2 - initialVideoHeight / 2;
  const morphDistance = Math.max(1, startTop - stickyTop);
  const afterBorderTop =
    Number.parseFloat(getComputedStyle(engineeringAfter).borderTopWidth) || 0;
  let normalVideoDocumentTop = afterBorderTop;
  for (
    let node = engineeringMorph;
    node instanceof HTMLElement;
    node = node.offsetParent
  ) {
    normalVideoDocumentTop += node.offsetTop;
  }
  const morphStartScroll = normalVideoDocumentTop - startTop;
  const travelled = window.scrollY - morphStartScroll;
  const progress = Math.max(0, Math.min(1, travelled / morphDistance));
  const eased =
    progress * progress * progress * (progress * (progress * 6 - 15) + 10);
  const currentAfterWidth =
    comparisonWidth - (comparisonWidth - afterWidth) * eased;
  const currentInlineHeight =
    initialVideoHeight - (initialVideoHeight - finalInlineHeight) * eased;

  engineeringMorph.style.setProperty(
    "--comparison-after-width",
    `${currentAfterWidth.toFixed(2)}px`,
  );
  engineeringMorph.style.setProperty(
    "--comparison-inline-height",
    `${currentInlineHeight.toFixed(2)}px`,
  );

  const beforeMediaHeight =
    beforeVideo?.offsetHeight || 0;
  const afterMediaHeight =
    engineeringAfter.querySelector(".inline-evidence-layout")?.offsetHeight || 0;
  const beforeCaptionHeight =
    engineeringBefore.querySelector("figcaption")?.offsetHeight || 0;
  const afterCaptionHeight =
    engineeringAfter.querySelector("figcaption")?.offsetHeight || 0;
  const cardHeight = Math.max(
    beforeMediaHeight + beforeCaptionHeight + 6,
    afterMediaHeight + afterCaptionHeight + 6,
  );
  engineeringMorph.style.setProperty(
    "--comparison-card-height",
    `${Math.ceil(cardHeight)}px`,
  );
  engineeringMorph.style.setProperty(
    "--comparison-stage-height",
    `${Math.ceil(cardHeight + 120)}px`,
  );

  const revealPosition = Math.max(
    0,
    Math.min(1, (progress - 0.12) / 0.68),
  );
  const revealProgress =
    revealPosition * revealPosition * (3 - 2 * revealPosition);
  const arrowPosition = Math.max(
    0,
    Math.min(1, (progress - 0.52) / 0.42),
  );
  const arrowOpacity =
    arrowPosition * arrowPosition * (3 - 2 * arrowPosition);
  engineeringMorph.style.setProperty(
    "--comparison-before-opacity",
    revealProgress.toFixed(4),
  );
  engineeringMorph.style.setProperty(
    "--comparison-arrow-opacity",
    arrowOpacity.toFixed(4),
  );
}

function scheduleEngineeringMorph() {
  if (engineeringMorphFrame) return;
  engineeringMorphFrame = window.requestAnimationFrame(updateEngineeringMorph);
}

function setFilmLanguage(language) {
  if (!film) return;
  const source = film.querySelector("source");
  const track = film.querySelector("track");
  const nextSource =
    language === "zh" ? film.dataset.srcZh : film.dataset.srcEn;
  const nextPoster =
    language === "zh" ? film.dataset.posterZh : film.dataset.posterEn;
  const nextLabel =
    language === "zh" ? film.dataset.labelZh : film.dataset.labelEn;

  if (nextLabel) film.setAttribute("aria-label", nextLabel);
  if (nextPoster) film.poster = nextPoster;
  if (!source || !nextSource || source.getAttribute("src") === nextSource) {
    if (track) {
      track.src =
        language === "zh"
          ? "../../assets/roborsi/demo/roborsi-demo-zh.vtt"
          : "../../assets/roborsi/demo/roborsi-demo-en.vtt";
      track.srclang = language === "zh" ? "zh-CN" : "en";
      track.label = language === "zh" ? "中文" : "English";
    }
    return;
  }

  const wasPaused = film.paused;
  const currentTime = Number.isFinite(film.currentTime) ? film.currentTime : 0;
  source.src = nextSource;
  if (track) {
    track.src =
      language === "zh"
        ? "../../assets/roborsi/demo/roborsi-demo-zh.vtt"
        : "../../assets/roborsi/demo/roborsi-demo-en.vtt";
    track.srclang = language === "zh" ? "zh-CN" : "en";
    track.label = language === "zh" ? "中文" : "English";
  }
  film.load();
  film.addEventListener(
    "loadedmetadata",
    () => {
      if (Number.isFinite(film.duration)) {
        film.currentTime = Math.min(currentTime, Math.max(0, film.duration - 0.1));
      }
      if (!wasPaused) film.play().catch(() => undefined);
    },
    { once: true },
  );
}

function frameworkVideoIsVisible(video) {
  if (getComputedStyle(video).display === "none") return false;
  const bounds = video.getBoundingClientRect();
  return bounds.bottom > 0 && bounds.top < window.innerHeight;
}

function syncFrameworkVideos() {
  frameworkVideos.forEach((video) => {
    if (
      !document.hidden &&
      !reduceMotion.matches &&
      frameworkVideoIsVisible(video)
    ) {
      video.play().catch(() => undefined);
    } else {
      video.pause();
    }
  });
}

function agentAnimationIsVisible(animation) {
  const bounds = animation.getBoundingClientRect();
  return bounds.bottom > 0 && bounds.top < window.innerHeight;
}

function syncAgentAnimations() {
  agentAnimations.forEach((animation) => {
    animation.classList.toggle(
      "is-running",
      !document.hidden &&
        !reduceMotion.matches &&
        agentAnimationIsVisible(animation),
    );
  });
}

function setSkillTreeStage(value, language = root.dataset.lang) {
  if (
    !skillTreeRange ||
    !skillTreeStage ||
    !skillTreeFrames.length ||
    !skillTreeIndex ||
    !skillTreeTitle ||
    !skillTreeCopy
  ) {
    return;
  }
  const position = Math.max(
    1,
    Math.min(skillTreeStages.length, Number(value) || 1),
  );
  const stageIndex = Math.round(position) - 1;
  skillTreeFrames.forEach((frame, index) => {
    frame.classList.toggle("is-active", index === stageIndex);
  });
  const stage = skillTreeStages[stageIndex];
  const localized = stage[language === "zh" ? "zh" : "en"];
  skillTreeStage.setAttribute(
    "aria-label",
    language === "zh"
      ? `TSR 第 ${stageIndex + 1} 轮：${localized.title}`
      : `Top-down Skill Refinement round ${stageIndex + 1}: ${localized.title}`,
  );
  skillTreeIndex.textContent =
    `ROUND ${String(stageIndex + 1).padStart(2, "0")} / 06`;
  skillTreeTitle.textContent = localized.title;
  skillTreeCopy.textContent = localized.copy;
  skillTreeRange.value = String(position);
  skillTreeRange.setAttribute(
    "aria-valuetext",
    language === "zh"
      ? `第 ${stageIndex + 1} 轮：${localized.title}`
      : `Round ${stageIndex + 1}: ${localized.title}`,
  );
  skillTreeRoundButtons.forEach((button, index) => {
    const active = index === stageIndex;
    button.classList.toggle("is-active", active);
    button.setAttribute("aria-pressed", String(active));
  });
  skillTreeRange.style.setProperty(
    "--skill-tree-progress",
    `${((position - 1) / (skillTreeStages.length - 1)) * 100}%`,
  );
  skillTreePlayback.value = position;
}

function radarCoordinates(values) {
  return values.map((value, index) => {
    const angle = -Math.PI / 2 + (index * Math.PI) / 3;
    const radius = 225 * (value / 100);
    return {
      x: 320 + radius * Math.cos(angle),
      y: 280 + radius * Math.sin(angle),
    };
  });
}

function setCapabilityRadarStage(value) {
  if (
    !capabilityRadarRange ||
    !capabilityRadarShape ||
    capabilityRadarPoints.length !== 6 ||
    capabilityRadarAxisValues.length !== 6 ||
    !capabilityRadarRound ||
    !capabilityRadarCenter
  ) {
    return;
  }
  const position = Math.max(1, Math.min(5, Number(value) || 1));
  const lower = Math.floor(position) - 1;
  const upper = Math.ceil(position) - 1;
  const blend = position - Math.floor(position);
  const profile = capabilityRadarProfiles[lower].map(
    (current, index) =>
      current +
      (capabilityRadarProfiles[upper][index] - current) * blend,
  );
  const rawProfile = capabilityRadarRawProfiles[lower].map(
    (current, index) =>
      current +
      (capabilityRadarRawProfiles[upper][index] - current) * blend,
  );
  const points = radarCoordinates(profile);
  capabilityRadarShape.setAttribute(
    "points",
    points.map((point) => `${point.x.toFixed(1)},${point.y.toFixed(1)}`).join(" "),
  );
  capabilityRadarPoints.forEach((point, index) => {
    point.setAttribute("cx", points[index].x.toFixed(1));
    point.setAttribute("cy", points[index].y.toFixed(1));
  });
  const lowerRound = lower + 1;
  const upperRound = upper + 1;
  const transition =
    lowerRound !== upperRound && blend >= 0.02 && blend <= 0.98;
  const round = blend > 0.98 ? upperRound : lowerRound;
  const isZh = root.dataset.lang === "zh";
  capabilityRadarAxisValues.forEach((label, index) => {
    const raw = rawProfile[index];
    let value = "";
    if (index === 0) value = `≤ ${(raw / 1_000_000).toFixed(1)}M`;
    if (index === 1) value = `≤ ${(raw / 3600).toFixed(1)} h`;
    if (index === 2) {
      value = isZh
        ? `${Math.round(raw)} / 6 轮`
        : `${Math.round(raw)} / 6 releases`;
    }
    if (index === 3) {
      value = isZh
        ? `≤ ${Math.round(raw).toLocaleString("zh-CN")} 次`
        : `≤ ${Math.round(raw).toLocaleString("en-US")} calls`;
    }
    if (index === 4 || index === 5) {
      value = `${Math.round(raw)} / 120`;
    }
    label.textContent = value;
  });
  capabilityRadarRound.textContent = transition
    ? `PASS ${lowerRound} → ${upperRound}`
    : `PASS ${round}`;
  capabilityRadarCenter.textContent = transition
    ? `ROUND ${lowerRound}–${upperRound}`
    : `ROUND ${round}`;
  capabilityRadarRange.value = String(position);
  capabilityRadarRange.setAttribute(
    "aria-valuetext",
    transition
      ? root.dataset.lang === "zh"
        ? `第 ${lowerRound} 轮到第 ${upperRound} 轮的过渡`
        : `Transition from Pass ${lowerRound} to Pass ${upperRound}`
      : root.dataset.lang === "zh"
        ? `第 ${round} 轮`
        : `Pass ${round}`,
  );
  capabilityRadarRange.style.setProperty(
    "--radar-progress",
    `${((position - 1) / 4) * 100}%`,
  );
  capabilityRadarPlayback.value = position;
}

function realRobotSkillTreeMaxTime() {
  if (
    !realRobotSkillTreeVideo ||
    !Number.isFinite(realRobotSkillTreeVideo.duration)
  ) {
    return 0;
  }
  return Math.min(
    Math.max(0, realRobotSkillTreeVideo.duration - 1 / 30),
    REAL_ROBOT_SKILL_TREE_KEYFRAMES.at(-1).time,
  );
}

function realRobotSkillTreeTimeForRound(round) {
  const bounded = Math.max(
    1,
    Math.min(REAL_ROBOT_SKILL_TREE_ROUNDS, Number(round) || 1),
  );
  const upperIndex = REAL_ROBOT_SKILL_TREE_KEYFRAMES.findIndex(
    (keyframe) => keyframe.round >= bounded,
  );
  if (upperIndex <= 0) return 0;
  const upper = REAL_ROBOT_SKILL_TREE_KEYFRAMES[upperIndex];
  const lower = REAL_ROBOT_SKILL_TREE_KEYFRAMES[upperIndex - 1];
  const progress = (bounded - lower.round) / (upper.round - lower.round);
  return lower.time + progress * (upper.time - lower.time);
}

function realRobotSkillTreeRoundForTime(time) {
  const maxTime = realRobotSkillTreeMaxTime();
  if (maxTime <= 0) return 1;
  const bounded = Math.max(0, Math.min(maxTime, Number(time) || 0));
  const upperIndex = REAL_ROBOT_SKILL_TREE_KEYFRAMES.findIndex(
    (keyframe) => keyframe.time >= bounded,
  );
  if (upperIndex <= 0) return 1;
  const upper = REAL_ROBOT_SKILL_TREE_KEYFRAMES[upperIndex];
  const lower = REAL_ROBOT_SKILL_TREE_KEYFRAMES[upperIndex - 1];
  const progress = (bounded - lower.time) / (upper.time - lower.time);
  return Math.max(
    1,
    Math.min(
      REAL_ROBOT_SKILL_TREE_ROUNDS,
      Math.floor(
        lower.round + progress * (upper.round - lower.round) + Number.EPSILON,
      ),
    ),
  );
}

function updateRealRobotSkillTreeRound(round, seek = false) {
  if (!realRobotSkillTreeRange || !realRobotSkillTreeRound) return;
  const bounded = Math.max(
    1,
    Math.min(REAL_ROBOT_SKILL_TREE_ROUNDS, Math.round(Number(round) || 1)),
  );
  realRobotSkillTreeRange.value = String(bounded);
  realRobotSkillTreeRound.textContent =
    `ROUND ${String(bounded).padStart(3, "0")} / 104`;
  realRobotSkillTreeRange.setAttribute(
    "aria-valuetext",
    root.dataset.lang === "zh"
      ? `第 ${bounded} 轮，共 104 轮`
      : `Round ${bounded} of 104`,
  );
  realRobotSkillTreeRange.style.setProperty(
    "--skill-tree-progress",
    `${((bounded - 1) / (REAL_ROBOT_SKILL_TREE_ROUNDS - 1)) * 100}%`,
  );
  if (
    seek &&
    realRobotSkillTreeVideo &&
    realRobotSkillTreeMaxTime() > 0
  ) {
    realRobotSkillTreeVideo.currentTime =
      realRobotSkillTreeTimeForRound(bounded);
  }
}

function setRoundPlayback(kind, playing) {
  const isSkillTree = kind === "skill-tree";
  const playback = isSkillTree
    ? skillTreePlayback
    : capabilityRadarPlayback;
  const minimum = 1;
  const maximum = isSkillTree ? skillTreeStages.length : 5;
  playback.playing = Boolean(playing) && !reduceMotion.matches;
  if (playback.playing) {
    if (playback.value >= maximum - 0.001) {
      playback.value = minimum;
      if (isSkillTree) setSkillTreeStage(minimum);
      else setCapabilityRadarStage(minimum);
    }
    playback.startTime =
      performance.now() -
      ((playback.value - minimum) / (maximum - minimum)) * 8000;
  } else {
    playback.startTime = null;
  }
  updateRoundPlayButton(kind);
}

function roundPlaybackPosition(now, playback, minimum, maximum) {
  if (playback.startTime === null) {
    playback.startTime =
      now -
      ((playback.value - minimum) / (maximum - minimum)) * 8000;
  }
  const cycle = 9400;
  const phase = (now - playback.startTime) % cycle;
  if (phase >= 8000) return maximum;
  return minimum + (maximum - minimum) * (phase / 8000);
}

function lockActiveNavigation(target) {
  lockedNavigation = {
    target: target.id,
    chapter: target.dataset.chapter,
  };
  lockedNavigationSeen = false;
  setActiveNavigation(lockedNavigation.target, lockedNavigation.chapter);
}

function maintainLockedNavigation() {
  if (!lockedNavigation) return false;
  const target = document.getElementById(lockedNavigation.target);
  if (!target) {
    lockedNavigation = null;
    lockedNavigationSeen = false;
    return false;
  }
  const rect = target.getBoundingClientRect();
  const visible = rect.bottom > 0 && rect.top < window.innerHeight;
  if (visible) lockedNavigationSeen = true;
  if (!visible && lockedNavigationSeen) {
    lockedNavigation = null;
    lockedNavigationSeen = false;
    return false;
  }
  setActiveNavigation(lockedNavigation.target, lockedNavigation.chapter);
  return true;
}

function isAtDocumentEnd() {
  return (
    window.scrollY + window.innerHeight >=
    document.documentElement.scrollHeight - 4
  );
}

function shuffled(items) {
  const next = [...items];
  for (let index = next.length - 1; index > 0; index -= 1) {
    const swap = Math.floor(Math.random() * (index + 1));
    [next[index], next[swap]] = [next[swap], next[index]];
  }
  return next;
}

function selectRandomSuccesses() {

    const descZh = document.createElement("span");
    descZh.className = "lang-zh";
    descZh.textContent = group.desc_zh;
    const descEn = document.createElement("span");
    descEn.className = "lang-en";
    descEn.textContent = group.desc_en;
    desc.append(descZh, descEn);
    const count = document.createElement("small");
    count.textContent = String(groupItems.length);
    header.append(title, desc, count);
    fragments.push(header);
    fragments.push(
      ...groupItems.map((item) =>
        createEpisodeCard(item, "random-success-card trace-enabled-card"),
      ),
    );
  });
  if (!fragments.length) return false;
  grid.replaceChildren(...fragments);
  grid.classList.add("is-grouped");
  document
    .querySelector("[data-randomize-successes]")
    ?.closest(".random-gallery-toolbar")
    ?.setAttribute("hidden", "");
  return true;
}

const SVG_NS = "http://www.w3.org/2000/svg";

function svgEl(name, attrs) {
  const el = document.createElementNS(SVG_NS, name);
  Object.keys(attrs || {}).forEach((key) => el.setAttribute(key, attrs[key]));
  return el;
}

function drawIterationCurve(svg, info) {
  if (!svg || !liberoIterationData?.curve?.length) return;
  const curve = liberoIterationData.curve;
  const total = liberoIterationData.total_tasks || 120;
  const left = 44;
  const right = 328;
  const top = 14;
  const bottom = 126;
  const relMin = curve[0].release;
  const relMax = curve[curve.length - 1].release;
  const x = (release) =>
    left + ((release - relMin) / (relMax - relMin || 1)) * (right - left);
  const y = (count) => bottom - (count / total) * (bottom - top);
  svg.replaceChildren();
  const grid = svgEl("g", { class: "iteration-grid" });
  [0.25, 0.5, 0.75].forEach((ratio) => {
    const gy = y(total * ratio);
    grid.append(
      svgEl("line", { x1: left, y1: gy, x2: right, y2: gy }),
    );
    const label = svgEl("text", { x: left - 6, y: gy + 4, "text-anchor": "end" });
    label.textContent = `${Math.round(ratio * 100)}%`;
    grid.append(label);
  });
  grid.append(
    svgEl("line", { x1: left, y1: bottom, x2: right, y2: bottom }),
  );
  const zero = svgEl("text", { x: left - 6, y: bottom + 4, "text-anchor": "end" });
  zero.textContent = "0%";
  grid.append(zero);
  const firstTick = svgEl("text", {
    x: left,
    y: bottom + 16,
    "text-anchor": "start",
    class: "iteration-x-label",
  });
  firstTick.textContent = `release ${relMin}`;
  const lastTick = svgEl("text", {
    x: right,
    y: bottom + 16,
    "text-anchor": "end",
    class: "iteration-x-label",
  });
  lastTick.textContent = `release ${relMax}`;
  grid.append(firstTick, lastTick);
  svg.append(grid);
  const path = curve
    .map((point, index) =>
      `${index === 0 ? "M" : "L"}${x(point.release).toFixed(1)} ${y(point.cumulative).toFixed(1)}`,
    )
    .join(" ");
  svg.append(svgEl("path", { class: "iteration-line", d: path }));
  curve.forEach((point) => {
    svg.append(
      svgEl("circle", {
        class: "iteration-point",
        cx: x(point.release).toFixed(1),
        cy: y(point.cumulative).toFixed(1),
        r: 2.4,
      }),
    );
  });
  const markX = x(info.release);
  const markY = y(info.cumulative);
  svg.append(
    svgEl("circle", {
      class: "iteration-mark",
      cx: markX.toFixed(1),
      cy: markY.toFixed(1),
      r: 5,
    }),
  );
  const mark = svgEl("text", {
    class: "iteration-mark-label",
    x: (markX + (markX > right - 70 ? -10 : 10)).toFixed(1),
    y: (markY - 8 < top + 8 ? markY + 16 : markY - 8).toFixed(1),
    "text-anchor": markX > right - 70 ? "end" : "start",
  });
  mark.textContent = `${info.cumulative}/${total}`;
  svg.append(mark);
}

function renderTraceIteration(item, isZh) {
  const block = traceDialog?.querySelector("[data-trace-iteration]");
  if (!block) return;
  const info = liberoIterationData?.items?.[item.id];
  if (!info || !liberoIterationData?.curve?.length) {
    block.setAttribute("hidden", "");
    return;
  }
  block.removeAttribute("hidden");
  const releaseLabel = block.querySelector("[data-trace-iteration-release]");
  if (releaseLabel) {
    releaseLabel.textContent = isZh
      ? `第 ${info.release_index}/${info.releases_total} 个迭代 release · release ${info.release} 首次通过`
      : `iteration ${info.release_index}/${info.releases_total} · first pass at release ${info.release}`;
  }
  drawIterationCurve(block.querySelector("[data-trace-iteration-curve]"), info);
}

        : "exact episode"

    item.verdict === "simulator_success"
      ? isZh
        ? "仿真成功"
        : "simulator success"
      : item.verdict;
  renderTraceIteration(item, isZh);
  const steps = traceDialog.querySelector("[data-trace-steps]");
  steps.replaceChildren(
    ...item.tool_chain.map((step) => {
      const row = document.createElement("li");
      row.className = step.ok ? "is-success" : "is-failure";
      const index = document.createElement("span");
      index.textContent = String(step.index ?? "");
      const body = document.createElement("div");
      const tool = document.createElement("strong");
      tool.textContent = step.tool;
      const status = document.createElement("small");
      status.textContent = step.ok
        ? isZh
          ? "通过"
          : "OK"

function resumeVisibleVideos() {
  evidenceVideos.forEach((video) => {
    const rect = video.getBoundingClientRect();
    const visible = rect.bottom > 0 && rect.top < window.innerHeight;
    if (visible && video.dataset.userPaused !== "true" && !reduceMotion.matches) {
      playVideo(video);
    }
  });
}

}

function syncVideoLabel(video) {
  const playing = !video.paused;
  const language = root.dataset.lang === "zh" ? "zh" : "en";
  const container = video.closest("article, aside, figure");
  const title =
    (language === "zh" ? video.dataset.titleZh : video.dataset.titleEn) ||
    container
      ?.querySelector(`h3.lang-${language}, h3 .lang-${language}`)
      ?.textContent?.trim() ||
    container
      ?.querySelector(`strong.lang-${language}`)
      ?.textContent?.trim() ||
    "";
  if (video.dataset.physicalVideo !== undefined) {
    const label =
      language === "zh"
        ? `打开${title || "真机"}视频，可展开交互式工具调用链`
        : `Open ${title || "real-robot"} video; interactive tool chain available`;
    video.setAttribute("aria-label", label);
    video.removeAttribute("aria-pressed");
    video.classList.toggle("is-paused", !playing);
    return;
  }
  if (video.dataset.traceId) {
    const label =
      language === "zh"
        ? `查看${title || "此回合"}工具调用链`
        : `View tool trace for ${title || "this episode"}`;
    video.setAttribute("aria-label", label);
    video.removeAttribute("aria-pressed");
    return;
  }
  const label =
    language === "zh"
      ? playing
        ? `暂停${title || "此"}视频`
        : `播放${title || "此"}视频`
      : playing
        ? `Pause ${title || "this"} video`
        : `Play ${title || "this"} video`;
  video.setAttribute("aria-label", label);
  video.setAttribute("aria-pressed", String(playing));
  video.classList.toggle("is-paused", !playing);
}

function pauseVideo(video, userInitiated = false) {
  video.pause();
  if (userInitiated) video.dataset.userPaused = "true";
  syncVideoLabel(video);
}

function playVideo(video, userInitiated = false) {
  if (userInitiated) delete video.dataset.userPaused;
  video.play().then(() => syncVideoLabel(video)).catch(() => syncVideoLabel(video));
}

function registerEvidenceVideo(video) {
  if (registeredVideos.has(video)) return;
  registeredVideos.add(video);
  evidenceVideos.push(video);
  video.setAttribute("role", "button");
  video.setAttribute("tabindex", "0");
  syncVideoLabel(video);
  const activate = () => {
    const item = episodeById.get(video.dataset.traceId);
    if (item) {
      openTrace(item);
      return;
    }
    if (video.paused) playVideo(video, true);
    else pauseVideo(video, true);
  };
  video.addEventListener("click", activate);
  video.addEventListener("keydown", (event) => {
    if (event.key !== "Enter" && event.key !== " ") return;
    event.preventDefault();
    activate();
  });
  videoObserver.observe(video);
}

const videoObserver = new IntersectionObserver(
  (entries) => {
    entries.forEach((entry) => {
      const video = entry.target;
      if (
        entry.isIntersecting &&
        !document.hidden &&
        !reduceMotion.matches &&
        video.dataset.userPaused !== "true"
      ) {
        playVideo(video);
      } else if (!entry.isIntersecting || document.hidden || reduceMotion.matches) {
        pauseVideo(video);
      }
    });
  },
  { rootMargin: "120px 0px", threshold: 0.15 },
);

async function loadSimulationShowcase() {
  const response = await fetch(
    "../../assets/roborsi/evidence/simulation-showcase-v1.json?v=20260902-1",
  );
  if (!response.ok) {
    throw new Error(`showcase manifest HTTP ${response.status}`);
  }
  const payload = await response.json();
  if (
    payload.schema !== "robot-rsi.simulation-showcase.v1" ||
    payload.counts?.act !== 1 ||
    payload.counts?.libero_showcase !== 9 ||
    payload.counts?.libero_pool !== 24 ||
    payload.counts?.robotwin !== 24
  ) {
    throw new Error("unexpected showcase manifest");
  }
  const robotwinShowcase = payload.robotwin_showcase.filter(
    (item) => item.showcase_eligible !== false,
  );
  if (robotwinShowcase.length < 6) {
    throw new Error("insufficient RoboTwin showcase episodes");
  }
  [
    ...payload.libero_showcase,
    ...payload.libero_random_pool,
    ...robotwinShowcase,
    ...payload.act_showcase,
  ].forEach((item) => episodeById.set(item.id, item));
  randomSuccessPool = payload.libero_random_pool;
  renderEpisodeGrid(
    "[data-libero-showcase]",
    payload.libero_showcase,
    "evidence-card trace-enabled-card",
  );
  robotwinSuccessPool = robotwinShowcase;
  codeShowcasePool = payload.libero_showcase;
  renderRandomRobotwin();
  renderRandomSuccesses();
  renderRandomCodeShowcase();
  setLanguage(root.dataset.lang);
}

  try {

  }
}

const copyButton = document.querySelector("[data-copy-bibtex]");
copyButton?.addEventListener("click", async () => {
  const bibtex = document.querySelector("#bibtex")?.textContent || "";
  if (!bibtex) return;
  try {
    let copied = false;
    if (navigator.clipboard?.writeText) {
      try {
        await navigator.clipboard.writeText(bibtex);
        copied = true;
      } catch {
        copied = false;
      }
    }
    if (!copied) {
      const fallbackInput = document.createElement("textarea");
      fallbackInput.value = bibtex;
      fallbackInput.setAttribute("readonly", "");
      fallbackInput.style.position = "fixed";
      fallbackInput.style.opacity = "0";
      document.body.appendChild(fallbackInput);
      fallbackInput.select();
      copied = document.execCommand("copy");
      fallbackInput.remove();
      if (!copied) throw new Error("BibTeX copy failed");
    }
    const originalZh = copyButton.querySelector(".lang-zh")?.textContent;
    const originalEn = copyButton.querySelector(".lang-en")?.textContent;
    if (copyButton.querySelector(".lang-zh")) {
      copyButton.querySelector(".lang-zh").textContent = "已复制";
    }
    if (copyButton.querySelector(".lang-en")) {
      copyButton.querySelector(".lang-en").textContent = "Copied";
    }
    window.setTimeout(() => {
      if (copyButton.querySelector(".lang-zh")) {
        copyButton.querySelector(".lang-zh").textContent = originalZh || "复制";
      }
      if (copyButton.querySelector(".lang-en")) {
        copyButton.querySelector(".lang-en").textContent = originalEn || "Copy";
      }
    }, 1600);
  } catch {
    document.querySelector("#bibtex")?.parentElement?.focus();
  }
});
